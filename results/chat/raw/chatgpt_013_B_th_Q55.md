WITH ProductRevenue AS (     SELECT         Category,         ProductName,         SUM(Revenue) AS TotalRevenue,         ROW_NUMBER() OVER (             PARTITION BY Category             ORDER BY SUM(Revenue) DESC, ProductName ASC         ) AS Rank     FROM Ecommerce     GROUP BY Category, ProductName ) SELECT     Category,     ProductName,     TotalRevenue FROM ProductRevenue WHERE Rank <= 3 ORDER BY Category ASC, Rank ASC;
